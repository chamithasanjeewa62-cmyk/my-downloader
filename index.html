/* Video URL input validation - CONNECTED TO VERCEL BACKEND */ const form=document.getElementById("downloadForm"); const input=document.getElementById("videoUrl"); const message=document.getElementById("message"); const btn = document.querySelector('.analyze-btn'); form.addEventListener("submit",async function(e){ e.preventDefault(); const value=input.value.trim(); try{ const url=new URL(value); if(url.protocol!=="https:" && url.protocol!=="http:") throw new Error("Invalid"); }catch{ message.classList.add("show"); message.textContent="Please enter a valid link beginning with https://"; return; } // Loading state const oldText = btn.innerHTML; btn.innerHTML = "Processing... ⏳"; btn.disabled = true; message.classList.add("show"); message.innerHTML="⏳ Analyzing...
Connecting to download server..."; try{ const res = await fetch('/api/download', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ url: value }) }); const data = await res.json(); console.log(data); if(data.url){ message.innerHTML = ` ✓ Ready to Download!

DOWNLOAD VIDEO - No Watermark 🔥
If not downloading, long press and Save Link `; } else { message.innerHTML = `Error
${JSON.stringify(data)}`; } }catch(err){ message.innerHTML = `Failed
${err.message}`; } btn.innerHTML = oldText; btn.disabled = false; });
